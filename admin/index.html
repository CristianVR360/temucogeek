Script SQL para Supabase

Ejecuta este código en el SQL Editor de tu consola Supabase para crear la tabla de postulaciones: 

CREATE TABLE IF NOT EXISTS postulaciones (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  nombre_expositor TEXT NOT NULL,
  nombre_marca TEXT NOT NULL,
  redes_sociales TEXT NOT NULL,
  telefono TEXT NOT NULL,
  email TEXT NOT NULL,
  ciudad TEXT NOT NULL,
  categorias TEXT NOT NULL,
  espacio_tipo TEXT NOT NULL,
  is_formalizado BOOLEAN DEFAULT false,
  rut_empresa TEXT,
  razon_social TEXT,
  giro_empresa TEXT,
  descripcion_productos TEXT NOT NULL,
  tipo_postulacion TEXT DEFAULT 'expositor',
  estado TEXT DEFAULT 'pendiente',
  es_canje BOOLEAN DEFAULT false,
  notas_internas TEXT
);

-- Agregar columnas faltantes en tablas existentes si no existen
ALTER TABLE postulaciones ADD COLUMN IF NOT EXISTS tipo_postulacion TEXT DEFAULT 'expositor';
ALTER TABLE postulaciones ADD COLUMN IF NOT EXISTS es_canje BOOLEAN DEFAULT false;
ALTER TABLE postulaciones_cosplay ADD COLUMN IF NOT EXISTS es_canje BOOLEAN DEFAULT false;

ALTER TABLE postulaciones ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Permitir inserciones públicas" ON postulaciones FOR INSERT WITH CHECK (true);
CREATE POLICY "Permitir lectura pública" ON postulaciones FOR SELECT USING (true);
CREATE POLICY "Permitir actualización pública" ON postulaciones FOR UPDATE USING (true);
CREATE POLICY "Permitir eliminación pública" ON postulaciones FOR DELETE USING (true);

-- Tabla Oficial para Concurso de Cosplay

  email TEXT NOT NULL,
  nombre_apoderado TEXT,
  rut_apoderado TEXT,

  origen TEXT NOT NULL,
  imagen_ref TEXT NOT NULL,
  observaciones TEXT,
  tipo_postulacion TEXT DEFAULT 'cosplay',
  estado TEXT DEFAULT 'pendiente',
  notas_internas TEXT

ALTER TABLE postulaciones_cosplay ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Permitir inserciones públicas cosplay" ON postulaciones_cosplay FOR INSERT WITH CHECK (true);
CREATE POLICY "Permitir lectura pública cosplay" ON postulaciones_cosplay FOR SELECT USING (true);
CREATE POLICY "Permitir actualización pública cosplay" ON postulaciones_cosplay FOR UPDATE USING (true);
CREATE POLICY "Permitir eliminación pública cosplay" ON postulaciones_cosplay FOR DELETE USING (true);

CREATE TABLE IF NOT EXISTS public.encuestas_cosplay (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  nota_organizacion INT CHECK (nota_organizacion BETWEEN 1 AND 10),
  nota_camerinos INT CHECK (nota_camerinos BETWEEN 1 AND 5),
  nota_ganadores INT CHECK (nota_ganadores BETWEEN 1 AND 10),
  justificacion_ganadores TEXT DEFAULT '',
  comentarios TEXT DEFAULT ''
);

CREATE TABLE IF NOT EXISTS public.encuestas_expositores (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  nombre_stand TEXT NOT NULL,
  nombre_contacto TEXT NOT NULL,
  email TEXT NOT NULL,
  telefono TEXT NOT NULL,
  tipo_stand TEXT DEFAULT 'Comercial',
  nota_logistica INT CHECK (nota_logistica BETWEEN 1 AND 5),
  nota_publico INT CHECK (nota_publico BETWEEN 1 AND 5),
  resultado_comercial TEXT DEFAULT 'Bueno',
  interes_proxima_edicion TEXT DEFAULT 'Sí, definitivamente',
  feedback_mejoras TEXT DEFAULT ''
);

CREATE TABLE IF NOT EXISTS public.encuestas_publico (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
  nombre_completo TEXT NOT NULL,
  rut TEXT NOT NULL,
  email TEXT NOT NULL,
  telefono TEXT NOT NULL,
  medio_difusion TEXT DEFAULT '',
  nota_general INT CHECK (nota_general BETWEEN 1 AND 10),
  zona_favorita TEXT DEFAULT 'Cosplay',
  nps_recomendacion INT CHECK (nps_recomendacion BETWEEN 1 AND 10),
  nota_instalaciones INT CHECK (nota_instalaciones BETWEEN 1 AND 5),
  nota_ingreso INT CHECK (nota_ingreso BETWEEN 1 AND 5),
  nota_comida INT CHECK (nota_comida BETWEEN 1 AND 5),
  nota_limpieza INT CHECK (nota_limpieza BETWEEN 1 AND 5),
  nota_tiendas INT CHECK (nota_tiendas BETWEEN 1 AND 5),
  precio_calidad TEXT DEFAULT '',
  aspecto_mejora TEXT DEFAULT '',
  sugerencias TEXT DEFAULT '',
  acepta_ley_datos BOOLEAN NOT NULL DEFAULT true,
  estado_entrada TEXT DEFAULT 'Pendiente'
);

ALTER TABLE public.encuestas_cosplay ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.encuestas_expositores ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.encuestas_publico ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Permitir insercion cosplay" ON public.encuestas_cosplay FOR ALL USING (true);
CREATE POLICY "Permitir insercion expositores" ON public.encuestas_expositores FOR ALL USING (true);
CREATE POLICY "Permitir insercion publico" ON public.encuestas_publico FOR ALL USING (true);
📧 Notificaciones Automáticas por Correo (Resend API & Supabase Webhooks)

El sitio envía correos automáticamente vía Resend API usando la clave RESEND_API_KEY. Configura un único Webhook en Supabase sobre la tabla postulaciones (todas las postulaciones, incluidos cosplayers, se registran ahí):

  1. Ve a tu consola de Supabase → Database → Webhooks.
  2. Verifica que exista solo 1 Webhook activo (elimina duplicados si existen).
  3. Name: notificar_postulacion_resend
  4. Table: postulaciones (Event: INSERT)
  5. Webhook Type: HTTP Request
  6. Method: POST → URL: https://www.temugeek.cl/api/send-email (o tu función backend)
  7. HTTP Headers:
    Authorization: Bearer RESEND_API_KEY
    Content-Type: application/json
